Tree & Graph Traversals Cheatsheet

Recursive call stacks, BFS queue level-batching, Kahn's in-degree topological sort, and Disjoint Set Union (DSU).

1. Tree Traversal Taxonomy

Order / Traversal Node Sequence Primary Use Case & Invariant Data Structure
Pre-Order (DFS) Root → Left → Right Tree cloning, prefix serialization, root-first decisions Call Stack / LIFO Stack
In-Order (DFS) Left → Root → Right Monotonically ascending order in Binary Search Trees Call Stack / LIFO Stack
Post-Order (DFS) Left → Right → Root Bottom-up aggregation (height, diameter, subtree sums) Call Stack / LIFO Stack
Level-Order (BFS) Level by level (top to bottom) Shortest path in unweighted trees, horizontal layer snapshots FIFO Queue (batch by size)

2. Graph Algorithms Complexity Crosswalk

Algorithm Graph Type Time Complexity Space Complexity Core Mechanism
Multi-Source BFS Unweighted Grid / Graph O(V + E) O(V) Seed queue with all sources at $t=0$; uniform wavefront expansion
Kahn's Algorithm Directed Graph (DAG) O(V + E) O(V) FIFO queue of nodes with in_degree == 0; cycle if completed $< V$
3-Color DFS Directed Graph O(V + E) O(V) White (0: unvisited), Gray (1: in stack), Black (2: done); Gray = cycle
Disjoint Set Union (DSU) Undirected Graph O(E × α(V)) O(V) Path compression + Union by rank; dynamic connectivity in nearly $O(1)$

# Template 1: Kahn's Topological Sort
from collections import deque

def topological_sort(v: int, edges: list[list[int]]) -> list[int]:
    adj = [[] for _ in range(v)]
    in_degree = [0] * v
    for dest, src in edges:
        adj[src].append(dest)
        in_degree[dest] += 1
    
    queue = deque([i for i in range(v) if in_degree[i] == 0])
    order = []
    while queue:
        u = queue.popleft()
        order.append(u)
        for nxt in adj[u]:
            in_degree[nxt] -= 1
            if in_degree[nxt] == 0:
                queue.append(nxt)
    return order if len(order) == v else []

# Template 2: Disjoint Set Union (DSU)
class DSU:
    def __init__(self, n: int):
        self.parent = list(range(n))
        self.rank = [0] * n

    def find(self, x: int) -> int:
        if self.parent[x] != x:
            self.parent[x] = self.find(self.parent[x]) # Path compression
        return self.parent[x]

    def union(self, x: int, y: int) -> bool:
        rx, ry = self.find(x), self.find(y)
        if rx == ry:
            return False # Cycle detected
        if self.rank[rx] < self.rank[ry]:
            rx, ry = ry, rx
        self.parent[ry] = rx
        if self.rank[rx] == self.rank[ry]:
            self.rank[rx] += 1
        return True
// Template 1: Kahn's Topological Sort
function topologicalSort(v: number, edges: number[][]): number[] {
  const adj: number[][] = Array.from({ length: v }, () => []);
  const inDegree = new Array(v).fill(0);
  for (const [dest, src] of edges) {
    adj[src].push(dest);
    inDegree[dest]++;
  }

  const queue: number[] = [];
  for (let i = 0; i < v; i++) {
    if (inDegree[i] === 0) queue.push(i);
  }

  const order: number[] = [];
  let head = 0;
  while (head < queue.length) {
    const u = queue[head++];
    order.push(u);
    for (const nxt of adj[u]) {
      if (--inDegree[nxt] === 0) queue.push(nxt);
    }
  }
  return order.length === v ? order : [];
}

// Template 2: Disjoint Set Union (DSU)
class DSU {
  private parent: Int32Array;
  private rank: Int32Array;

  constructor(n: number) {
    this.parent = new Int32Array(n);
    this.rank = new Int32Array(n);
    for (let i = 0; i < n; i++) this.parent[i] = i;
  }

  find(x: number): number {
    if (this.parent[x] !== x) {
      this.parent[x] = this.find(this.parent[x]); // Path compression
    }
    return this.parent[x];
  }

  union(x: number, y: number): boolean {
    let rx = this.find(x);
    let ry = this.find(y);
    if (rx === ry) return false;
    if (this.rank[rx] < this.rank[ry]) {
      const temp = rx; rx = ry; ry = temp;
    }
    this.parent[ry] = rx;
    if (this.rank[rx] === this.rank[ry]) this.rank[rx]++;
    return true;
  }
}
// Template 1: Kahn's Topological Sort
import java.util.*;

public int[] topologicalSort(int v, int[][] edges) {
    List<List<Integer>> adj = new ArrayList<>();
    for (int i = 0; i < v; i++) adj.add(new ArrayList<>());
    int[] inDegree = new int[v];

    for (int[] edge : edges) {
        adj.get(edge[1]).add(edge[0]);
        inDegree[edge[0]]++;
    }

    Queue<Integer> queue = new ArrayDeque<>();
    for (int i = 0; i < v; i++) {
        if (inDegree[i] == 0) queue.offer(i);
    }

    int[] order = new int[v];
    int idx = 0;
    while (!queue.isEmpty()) {
        int u = queue.poll();
        order[idx++] = u;
        for (int nxt : adj.get(u)) {
            if (--inDegree[nxt] == 0) queue.offer(nxt);
        }
    }
    return idx == v ? order : new int[]{};
}

public static class DSU {
    int[] parent;
    int[] rank;

    public DSU(int n) {
        parent = new int[n];
        rank = new int[n];
        for (int i = 0; i < n; i++) parent[i] = i;
    }

    public int find(int x) {
        if (parent[x] != x) {
            parent[x] = find(parent[x]); // Path compression
        }
        return parent[x];
    }

    public boolean union(int x, int y) {
        int rx = find(x), ry = find(y);
        if (rx == ry) return false;
        if (rank[rx] < rank[ry]) {
            int t = rx; rx = ry; ry = t;
        }
        parent[ry] = rx;
        if (rank[rx] == rank[ry]) rank[rx]++;
        return true;
    }
}
