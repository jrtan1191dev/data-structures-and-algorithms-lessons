Trie & Prefix Tree Cheatsheet

26-ary tree node pointer architecture, prefix querying mechanics, wildcard depth-first searches, and exponential multi-string grid pruning.

1. Trie vs. Hash Table Comparison

Dimension Trie (Prefix Tree) Hash Table (HashSet/Map)
Exact Word Search O(L) step-by-step pointer traversal O(L) average hashing + comparison
Prefix Search (startsWith) O(L) immediate termination at prefix node O(N × L) scan through all $N$ dictionary keys
Wildcard Matching (.) Recursive branch across non-null children Impractical; requires scanning all stored words
Memory Overhead High pointer overhead (26 pointers per node) Low; compact contiguous bucket array

2. Trie Operations Invariant Protocol

Standard Trie Operations ($L$ = word length)

  • Insert (word): Start at root. For each char c: if curr.children[c] is null, create new TrieNode. Advance curr = curr.children[c]. Mark curr.is_end = true. Time: $O(L)$, Space: $O(L)$.
  • Search (word): Walk matching child pointers. Return true if last node has is_end == true, else false. Time: $O(L)$, Space: $O(1)$.
  • StartsWith (prefix): Walk matching child pointers. If all characters are found, return true immediately without checking is_end. Time: $O(L)$, Space: $O(1)$.

3. Production Templates (Python • TypeScript • Java)

class TrieNode:
    def __init__(self):
        self.children = {}
        self.is_end = False

class Trie:
    def __init__(self):
        self.root = TrieNode()

    def insert(self, word: str) -> None:
        curr = self.root
        for ch in word:
            if ch not in curr.children:
                curr.children[ch] = TrieNode()
            curr = curr.children[ch]
        curr.is_end = True

    def search(self, word: str) -> bool:
        curr = self.root
        for ch in word:
            if ch not in curr.children:
                return False
            curr = curr.children[ch]
        return curr.is_end

    def starts_with(self, prefix: str) -> bool:
        curr = self.root
        for ch in prefix:
            if ch not in curr.children:
                return False
            curr = curr.children[ch]
        return True
class TrieNode {
  children: Array = new Array(26).fill(null);
  isEnd = false;
}

class Trie {
  private root = new TrieNode();

  insert(word: string): void {
    let curr = this.root;
    for (let i = 0; i < word.length; i++) {
      const idx = word.charCodeAt(i) - 97;
      if (curr.children[idx] === null) curr.children[idx] = new TrieNode();
      curr = curr.children[idx]!;
    }
    curr.isEnd = true;
  }

  search(word: string): boolean {
    let curr = this.root;
    for (let i = 0; i < word.length; i++) {
      const idx = word.charCodeAt(i) - 97;
      if (curr.children[idx] === null) return false;
      curr = curr.children[idx]!;
    }
    return curr.isEnd;
  }

  startsWith(prefix: string): boolean {
    let curr = this.root;
    for (let i = 0; i < prefix.length; i++) {
      const idx = prefix.charCodeAt(i) - 97;
      if (curr.children[idx] === null) return false;
      curr = curr.children[idx]!;
    }
    return true;
  }
}
public class Trie {
    private static class TrieNode {
        TrieNode[] children = new TrieNode[26];
        boolean isEnd = false;
    }

    private final TrieNode root = new TrieNode();

    public void insert(String word) {
        TrieNode curr = root;
        for (int i = 0; i < word.length(); i++) {
            int idx = word.charAt(i) - 'a';
            if (curr.children[idx] == null) {
                curr.children[idx] = new TrieNode();
            }
            curr = curr.children[idx];
        }
        curr.isEnd = true;
    }

    public boolean search(String word) {
        TrieNode curr = root;
        for (int i = 0; i < word.length(); i++) {
            int idx = word.charAt(i) - 'a';
            if (curr.children[idx] == null) return false;
            curr = curr.children[idx];
        }
        return curr.isEnd;
    }

    public boolean startsWith(String prefix) {
        TrieNode curr = root;
        for (int i = 0; i < prefix.length(); i++) {
            int idx = prefix.charAt(i) - 'a';
            if (curr.children[idx] == null) return false;
            curr = curr.children[idx];
        }
        return true;
    }
}
